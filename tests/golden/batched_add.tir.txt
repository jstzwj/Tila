func @batched_add(a: Buffer<f32, (M,N)>, b: Buffer<f32, (M,N)>, c: Buffer<f32, (M,N)>, M: Scalar(i32), N: Scalar(i32), BM: Constexpr(i32)=64, BN: Constexpr(i32)=128)
L0 = identity(64)
L1 = identity(128)
L2 = lift(L0, 1)
L3 = lift(L1, 0)
L4 = product(L0,L1)

%pid_n = program_id 1 : Scalar(i32) [#pid_n]
%BM = constexpr_ref BM : Scalar(i32)
%t0 = mul %pid_m %BM : Scalar(i32)
%r0 = arange 0 BM : Tile<i32, (64,), L0>
%rows = add %t0 %r0 : Tile<i32, (64,), L0> [#rows]
%BN = constexpr_ref BN : Scalar(i32)
%t1 = mul %pid_n %BN : Scalar(i32)
%r1 = arange 0 BN : Tile<i32, (128,), L1>
%cols = add %t1 %r1 : Tile<i32, (128,), L1> [#cols]
%rows2 = expand_dim %rows 1 : Tile<i32, (64,1), L2> [#rows2]
%cols2 = expand_dim %cols 0 : Tile<i32, (1,128), L3> [#cols2]

%N = sym_ref N : Scalar(i32)

%mask = and %m0 %m1 : Tile<bool, (64,128), L4> [#mask]
%p0 = addptr %a [%rows2, %cols2] : Address<f32, (64,128), L4>
%x = load %p0 mask=%mask : Tile<f32, (64,128), L4> [#x]
%p1 = addptr %b [%rows2, %cols2] : Address<f32, (64,128), L4>

%p2 = addptr %c [%rows2, %cols2] : Address<f32, (64,128), L4>
store %p2 %z mask=%mask
